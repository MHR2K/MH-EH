## 技术背景

### Jaro-Winkler 算法简介

Jaro-Winkler 相似度算法用于计算两个字符串的相似程度，特别适合短字符串比较：

- **返回值**: 0.0（完全不同）到 1.0（完全相同）
- **优势**: 对前缀匹配给予更高权重，适合搜索场景
- **应用场景**: 模糊搜索、拼写检查、数据去重

**算法原理**:
```
JaroWinklerSimilarity = JaroSimilarity + (prefix_length * scaling_factor * (1 - JaroSimilarity))
```

其中：
- `prefix_length`: 公共前缀长度（最多 4 个字符）
- `scaling_factor`: 权重因子（通常为 0.1）

public static boolean judgeSuitableTitle(@NonNull GalleryInfo gi, @NonNull String key, boolean fuzzySearch, boolean ignoreCase) {
    if (fuzzySearch) {
        String title = gi.title == null ? "" : gi.title;
        String titleJpn = gi.titleJpn == null ? "" : gi.titleJpn;
        if (ignoreCase) {
            title = title.toLowerCase();
            titleJpn = titleJpn.toLowerCase();
            key = key.toLowerCase();
        }
        return fuzzyContains(title, key, 0.7) || fuzzyContains(titleJpn, key, 0.7);
    }
    // ...原有精确匹配逻辑
}

public static boolean fuzzyContains(String text, String query, double threshold) {
    if (text.contains(query)) {
        return true;
    }
    double maxSimilarity = 0.0;
    for (int i = 0; i <= text.length() - query.length(); i++) {
        String substring = text.substring(i, i + query.length());
        double similarity = calculateJaroWinklerSimilarity(substring, query);
        maxSimilarity = Math.max(maxSimilarity, similarity);
        if (maxSimilarity >= threshold) {
            return true;
        }
    }
    return false;
}

public static double calculateJaroWinklerSimilarity(String s1, String s2) {
    if (s1.equals(s2)) {
        return 1.0;
    }
    int[] matches = getMatches(s1, s2);
    double m = matches[0];
    if (m == 0) {
        return 0.0;
    }
    double t = matches[1] / 2.0;
    double jaro = (m / s1.length() + m / s2.length() + (m - t) / m) / 3.0;
    int prefix = Math.min(matches[2], 4);
    return jaro + prefix * 0.1 * (1 - jaro);
}

private static int[] getMatches(String s1, String s2) {
    String max, min;
    if (s1.length() > s2.length()) {
        max = s1;
        min = s2;
    } else {
        max = s2;
        min = s1;
    }
    int range = Math.max(max.length() / 2 - 1, 0);
    int[] minIndexes = new int[min.length()];
    Arrays.fill(minIndexes, -1);
    boolean[] maxFlags = new boolean[max.length()];
    int matches = 0;
    for (int mi = 0; mi < min.length(); mi++) {
        char c1 = min.charAt(mi);
        for (int xi = Math.max(mi - range, 0); xi < Math.min(mi + range + 1, max.length()); xi++) {
            if (!maxFlags[xi] && c1 == max.charAt(xi)) {
                minIndexes[mi] = xi;
                maxFlags[xi] = true;
                matches++;
                break;
            }
        }
    }
    char[] ms1 = new char[matches];
    char[] ms2 = new char[matches];
    for (int i = 0, si = 0; i < min.length(); i++) {
        if (minIndexes[i] != -1) {
            ms1[si] = min.charAt(i);
            si++;
        }
    }
    for (int i = 0, si = 0; i < max.length(); i++) {
        if (maxFlags[i]) {
            ms2[si] = max.charAt(i);
            si++;
        }
    }
    int transpositions = 0;
    for (int mi = 0; mi < ms1.length; mi++) {
        if (ms1[mi] != ms2[mi]) {
            transpositions++;
        }
    }
    return new int[]{matches, transpositions, commonPrefix(s1, s2)};
}
```

@JvmStatic
fun judgeSuitableTitle(gi: GalleryInfo, key: String): Boolean {
    return judgeSuitableTitle(gi, key, Settings.getSearchFuzzy(), Settings.getSearchIgnoreCase())
}

@JvmStatic
fun judgeSuitableTitle(gi: GalleryInfo, key: String, fuzzySearch: Boolean, ignoreCase: Boolean): Boolean {
    var keyToSearch = key
    if (fuzzySearch) {
        var title = gi.title ?: ""
        var titleJpn = gi.titleJpn ?: ""
        if (ignoreCase) {
            title = title.lowercase()
            titleJpn = titleJpn.lowercase()
            keyToSearch = keyToSearch.lowercase()
        }
        return fuzzyContains(title, keyToSearch, 0.7) || fuzzyContains(titleJpn, keyToSearch, 0.7)
    }
    
    if (ignoreCase) {
        keyToSearch = keyToSearch.lowercase()
    }
    val title = gi.title?.let { if (ignoreCase) it.lowercase() else it } ?: return false
    val titleJpn = gi.titleJpn?.let { if (ignoreCase) it.lowercase() else it }
    return title.contains(keyToSearch) || titleJpn?.contains(keyToSearch) == true
}

@JvmStatic
fun fuzzyContains(text: String, query: String, threshold: Double): Boolean {
    if (text.contains(query)) {
        return true
    }
    
    var maxSimilarity = 0.0
    for (i in 0..(text.length - query.length)) {
        val substring = text.substring(i, i + query.length)
        val similarity = calculateJaroWinklerSimilarity(substring, query)
        maxSimilarity = max(maxSimilarity, similarity)
        if (maxSimilarity >= threshold) {
            return true
        }
    }
    return false
}

@JvmStatic
fun calculateJaroWinklerSimilarity(s1: String, s2: String): Double {
    if (s1 == s2) {
        return 1.0
    }
    
    val matches = getMatches(s1, s2)
    val m = matches[0].toDouble()
    if (m == 0.0) {
        return 0.0
    }
    
    val t = matches[1] / 2.0
    val jaro = (m / s1.length + m / s2.length + (m - t) / m) / 3.0
    val prefix = min(matches[2], 4)
    return jaro + prefix * 0.1 * (1 - jaro)
}

private fun getMatches(s1: String, s2: String): IntArray {
    val max: String
    val min: String
    if (s1.length > s2.length) {
        max = s1
        min = s2
    } else {
        max = s2
        min = s1
    }
    
    val range = max(max.length / 2 - 1, 0)
    val minIndexes = IntArray(min.length) { -1 }
    val maxFlags = BooleanArray(max.length)
    var matches = 0
    
    for (mi in min.indices) {
        val c1 = min[mi]
        for (xi in max(mi - range, 0) until min(mi + range + 1, max.length)) {
            if (!maxFlags[xi] && c1 == max[xi]) {
                minIndexes[mi] = xi
                maxFlags[xi] = true
                matches++
                break
            }
        }
    }
    
    val ms1 = CharArray(matches)
    val ms2 = CharArray(matches)
    var si = 0
    for (i in min.indices) {
        if (minIndexes[i] != -1) {
            ms1[si] = min[i]
            si++
        }
    }
    
    si = 0
    for (i in max.indices) {
        if (maxFlags[i]) {
            ms2[si] = max[i]
            si++
        }
    }
    
    var transpositions = 0
    for (mi in ms1.indices) {
        if (ms1[mi] != ms2[mi]) {
            transpositions++
        }
    }
    
    return intArrayOf(matches, transpositions, commonPrefix(s1, s2))
}
```

#### 3. `fuzzyContains(text, query, threshold)` - 模糊匹配核心

```kotlin
@JvmStatic
fun fuzzyContains(text: String, query: String, threshold: Double): Boolean
```

**功能**: 判断文本中是否包含与查询字符串相似的子串  
**参数**:
- `text`: 待搜索的文本
- `query`: 查询字符串
- `threshold`: 相似度阈值（0.0-1.0），推荐 0.7

**算法流程**:
1. **快速路径**: 如果精确包含，直接返回 `true`
2. **滑动窗口**: 遍历所有长度等于 `query.length()` 的子串
3. **相似度计算**: 对每个子串计算 Jaro-Winkler 相似度
4. **阈值判断**: 一旦找到相似度 ≥ threshold 的子串，返回 `true`

**示例**:
```kotlin
fuzzyContains("我的英雄学院", "英雄", 0.7)  // true (精确匹配)
fuzzyContains("我的英雄学院", "英隹", 0.7)  // true (相似度 ≈ 0.75)
fuzzyContains("我的英雄学院", "火影", 0.7)  // false (相似度 < 0.7)
```

---

#### 4. `calculateJaroWinklerSimilarity(s1, s2)` - 相似度计算

```kotlin
@JvmStatic
fun calculateJaroWinklerSimilarity(s1: String, s2: String): Double
```

**功能**: 计算两个字符串的 Jaro-Winkler 相似度  
**返回值**: `0.0`（完全不同）到 `1.0`（完全相同）

**算法原理**:
```
JaroWinklerSimilarity = JaroSimilarity + prefix_bonus

其中:
JaroSimilarity = (m/|s1| + m/|s2| + (m-t)/m) / 3
prefix_bonus = p × scaling_factor × (1 - JaroSimilarity)

m = 匹配字符数
t = 换位数
p = 公共前缀长度（最多 4）
scaling_factor = 0.1
```

**特性**:
- 对前缀匹配给予更高权重（适合搜索场景）
- 对字符顺序不敏感（适合处理拼写错误）
- 计算效率高（O(n×m)）

**示例**:
```kotlin
calculateJaroWinklerSimilarity("英雄", "英隹") // ≈ 0.75
calculateJaroWinklerSimilarity("MARTHA", "MARHTA") // ≈ 0.96
calculateJaroWinklerSimilarity("hello", "world") // ≈ 0.47
```

---

#### 5. `getMatches(s1, s2)` - 算法辅助方法

```kotlin
private fun getMatches(s1: String, s2: String): IntArray
```

**功能**: 计算 Jaro-Winkler 算法所需的三个关键指标  
**返回值**: `IntArray(3)` - `[matches, transpositions, prefix]`

**计算逻辑**:

1. **匹配窗口** (match window):
   ```kotlin
   range = max(max.length / 2 - 1, 0)
   ```
   只有在窗口范围内的字符才能被视为匹配

2. **匹配字符** (matches):
   - 两个字符在匹配窗口内
   - 字符相同
   - 每个字符只能匹配一次

3. **换位数** (transpositions):
   - 统计匹配字符中位置不同的数量
   - 用于惩罚字符顺序差异

4. **公共前缀** (prefix):
   - 从字符串开头开始连续相同的字符数
   - 最多计算 4 个字符

**示例**:
```kotlin
getMatches("英雄", "英隹")
// 返回: [2, 0, 1]
// matches=2 (两个字符都匹配)
// transpositions=0 (顺序相同)
// prefix=1 (第一个字符'英'相同)
```

## 已知问题与限制

### 1. 中文相似度偏低

**问题**: Jaro-Winkler 算法基于字符匹配，对中文汉字的语义相似性（如"钢"vs"鋼"）支持有限

**影响**: 简繁体混搜可能需要更低的阈值

**解决方案**: 
- 配合原有的 `ChineseConvertUtils` 进行简繁转换
- 考虑将阈值调整为可配置项

### 2. 性能开销

**问题**: 模糊搜索比精确搜索慢约 8 倍

**影响**: 大量下载项（> 5000）时可能有明显延迟

**优化方向**:
- 使用协程异步搜索
- 添加搜索结果缓存
- 实现增量搜索

### 3. 阈值固定

**问题**: 阈值硬编码为 0.7，无法根据场景调整

**建议**: 将阈值作为设置项暴露给用户
